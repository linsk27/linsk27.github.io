Buffer（缓冲器）

概念

  • 一个类似于Array的对象，即表示一段固定长度的内存空间，用于处理二进制数据

特点

  • Buffer大小固定且无法调整
  • 性能较好，直接对计算机内存进行操作
  • 每个元素的大小为1字节（byte），即8个bit

  1. 流式写入 *

程序打开一个文件是需要消耗资源的，流式写入可以减少打开关闭文件的次数。

流式写入方式适用于大文件写入或者频繁写入的场景，writeFile适合于写入频率较低的场景 如

  • 方案 1【分片上传，多请求并发】：前端slice切文件，并发发起多个请求上传分片；后端接收每一片，流式写入临时分片。优点：断网只重传失败分片，支持断点续传；缺点：请求多，要做并发控制。

  • 方案 2【单请求 body 流式上传】：前端把文件作为请求 body 的流，单个 HTTP 请求源源不断推送数据，后端同步流式写入磁盘。缺点：一旦网络中断，本次上传全部作废，无法断点续传。

const fs = require('fs')
const path = require('path')
const basePath = __dirname

const files = fs.readdirSync(basePath)
let fileList = []

try {
    files.forEach(item => {
        let data = item.split('-')
        let [numStr, ...rest] = data
        let num = Number(numStr)

        // 不是【数字-xxx】格式，直接跳过
        if (isNaN(num)) return

        let name = rest.join('-')
        fileList.push({ num, name, item })
    })

    // 按数字从小到大排序，对象捆绑，不会错位
    fileList.sort((a, b) => a.num - b.num)

    // 计算最大序号的位数，用来补0
    const maxNum = fileList[fileList.length - 1].num
    const maxLen = String(maxNum).length

    // 生成所有新文件名
    const result = fileList.map(obj => {
        const padded = String(obj.num).padStart(maxLen, '0')
        return `${padded}-${obj.name}`
    })

    // ==========【试运行预览模式】只打印，不修改任何文件！！==========
    result.forEach((item, index) => {
        console.log(`预览：${fileList[index].item}  →  ${item}`)
    })

    // ==========【真正重命名代码，预览确认无误后，再取消下面注释！】==========
    result.forEach((item, index) => {
        const oldPath = path.join(basePath, fileList[index].item)
        const newPath = path.join(basePath, item)
        fs.renameSync(oldPath, newPath)
        console.log(`已重命名：${fileList[index].item} → ${item}`)
    })
} catch (err) {
    console.log('出现错误：')
    console.log(err)
}

const express = require('express')
const path = require('path')
const app = express()

// 防盗链中间件
app.use((req, res, next) => {
    // 检测请求头中的referer 是否为127.0.0.1
    let referer = req.get('referer')
    console.log('[ referer ] >', referer)
    if (referer) {
        const url = new URL(referer)
        let hostname = url.hostname
        if (hostname !== '127.0.0.1') {
            res.status(404).send('<h1>404 NOT FOUND</h1>')
            return
        }
    }
    next()
})

var express = require('express')
var router = express.Router()

const formidable = require('formidable')

/* GET home page. */
router.get('/', function (req, res, next) {
    res.render('index', { title: 'Express' })
})

//显示网页的表单
router.get('/portrait', function (req, res) {
    res.render('portrait')
})
router.post('/portrait', function (req, res) {
    //创建form对象
    const form = formidable({
        multiples: true,
        // 设置上传文件的保存目录
        uploadDir: __dirname + '/../public/images',
        // 保存文件后缀
        keepExtensions: true
    })
    // 解析请求报文
    form.parse(req, (err, fields, files) => {
        if (err) {
            next(err)
            return
        }
        // console.log('[ fields ] >', fields) // 保存非文件数据的字段   {username:'abc'}
        // console.log('[ files ] >', files) // 保存文件数据的字段
        // 服务器保存该图片的访问url
        // /images/bzu57fhd4mc7t2x16kx294qz5.jpg
        let url = '/images/' + files.portrait.newFilename // 将来将此数据保存在数据库中
        res.send(url)
    })
})

module.exports = router
