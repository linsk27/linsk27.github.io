Buffer（缓冲器）

概念

  • 一个类似于Array的对象，即表示一段固定长度的内存空间，用于处理二进制数据

特点

  • Buffer大小固定且无法调整
  • 性能较好，直接对计算机内存进行操作
  • 每个元素的大小为1字节（byte），即8个bit

  1. 流式写入 *

程序打开一个文件是需要消耗资源的，流式写入可以减少打开关闭文件的次数。

流式写入方式适用于大文件写入或者频繁写入的场景，writeFile适合于写入频率较低的场景 如

  • 方案 1【分片上传，多请求并发】：前端slice切文件，并发发起多个请求上传分片；后端接收每一片，流式写入临时分片。优点：断网只重传失败分片，支持断点续传；缺点：请求多，要做并发控制。

  • 方案 2【单请求 body 流式上传】：前端把文件作为请求 body 的流，单个 HTTP 请求源源不断推送数据，后端同步流式写入磁盘。缺点：一旦网络中断，本次上传全部作废，无法断点续传。

const ws = fs.createWriteStream('../资料/笑看风云流式复制.mp4')
// 创建读取流对象
const rs = fs.createReadStream('../资料/笑看风云.mp4')

// // 为读取流绑定data事件
// rs.on('data', chunk => {

// })
// rs.on('end', () => {
//     //检查内存使用量 rss字段

// 更好的写法，自动处理背压，因为纯手写没处理背压。读得比写得快，数据块排队堆内存，rss 升高

const fs = require('fs')
const path = require('path')
const basePath = __dirname

const files = fs.readdirSync(basePath)
let fileList = []

try {
    files.forEach(item => {
        let data = item.split('-')
        let [numStr, ...rest] = data
        let num = Number(numStr)

        // 不是【数字-xxx】格式，直接跳过
        if (isNaN(num)) return

        let name = rest.join('-')
        fileList.push({ num, name, item })
    })

    // 按数字从小到大排序，对象捆绑，不会错位
    fileList.sort((a, b) => a.num - b.num)

    // 计算最大序号的位数，用来补0
    const maxNum = fileList[fileList.length - 1].num
    const maxLen = String(maxNum).length

    // 生成所有新文件名
    const result = fileList.map(obj => {
        const padded = String(obj.num).padStart(maxLen, '0')
        return `${padded}-${obj.name}`
    })

    // ==========【试运行预览模式】只打印，不修改任何文件！！==========
    result.forEach((item, index) => {
        console.log(`预览：${fileList[index].item}  →  ${item}`)
    })

    // ==========【真正重命名代码，预览确认无误后，再取消下面注释！】==========
    result.forEach((item, index) => {
        const oldPath = path.join(basePath, fileList[index].item)
        const newPath = path.join(basePath, item)
        fs.renameSync(oldPath, newPath)
        console.log(`已重命名：${fileList[index].item} → ${item}`)
    })
} catch (err) {
    console.log('出现错误：')
    console.log(err)
}
